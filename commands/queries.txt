29. INFINITE LOOP PREVENTION — History Truncation (cap at 20 turns)
    Expected: only the last 20 turns of history are passed to the LLM,
    preventing context window overflow on long sessions.
-------------------------------------------------------------

[Python test — verify history is trimmed]
python3 -c "
from app.agents.runner import AgentRunner
print('MAX_HISTORY:', AgentRunner._MAX_HISTORY)  # 20

# Simulate a long session
history = [{'role': 'user' if i % 2 == 0 else 'assistant', 'content': f'message {i}'} for i in range(50)]
print('Original history length:', len(history))

# Apply the guard (same as runner.run() does)
if len(history) > AgentRunner._MAX_HISTORY:
    history = history[-AgentRunner._MAX_HISTORY:]
print('Trimmed history length:', len(history))  # should be 20
print('First message after trim:', history[0]['content'])  # should be 'message 30'
"
